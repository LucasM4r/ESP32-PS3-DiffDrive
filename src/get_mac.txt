#include <WiFi.h>
#include <esp_wifi.h>
#include <esp_system.h>

void setup() {
  Serial.begin(115200);
  delay(1000);
  
  Serial.println("=== ESP32 MAC Address Information ===");
  
  // Obter MAC Address da interface WiFi Station
  uint8_t mac[6];
  esp_read_mac(mac, ESP_MAC_WIFI_STA);
  
  Serial.print("WiFi Station MAC: ");
  for(int i = 0; i < 6; i++) {
    Serial.printf("%02X", mac[i]);
    if(i < 5) Serial.print(":");
  }
  Serial.println();
  
  // Obter MAC Address da interface WiFi AP
  esp_read_mac(mac, ESP_MAC_WIFI_SOFTAP);
  
  Serial.print("WiFi AP MAC: ");
  for(int i = 0; i < 6; i++) {
    Serial.printf("%02X", mac[i]);
    if(i < 5) Serial.print(":");
  }
  Serial.println();
  
  // Obter MAC Address da interface Bluetooth
  esp_read_mac(mac, ESP_MAC_BT);
  
  Serial.print("Bluetooth MAC: ");
  for(int i = 0; i < 6; i++) {
    Serial.printf("%02X", mac[i]);
    if(i < 5) Serial.print(":");
  }
  Serial.println();
  
  // Método alternativo usando WiFi.macAddress()
  WiFi.mode(WIFI_MODE_STA);
  String macStr = WiFi.macAddress();
  Serial.print("WiFi MAC (String): ");
  Serial.println(macStr);
  
  Serial.println("=================================");
}

void loop() {
  // Nada no loop principal
  delay(10000);
}

// Função auxiliar para obter MAC como string
String getMacAddress() {
  uint8_t mac[6];
  esp_read_mac(mac, ESP_MAC_WIFI_STA);
  
  char macStr[18];
  sprintf(macStr, "%02X:%02X:%02X:%02X:%02X:%02X", 
          mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  
  return String(macStr);
}

// Função para obter MAC sem separadores
String getMacAddressPlain() {
  uint8_t mac[6];
  esp_read_mac(mac, ESP_MAC_WIFI_STA);
  
  char macStr[13];
  sprintf(macStr, "%02X%02X%02X%02X%02X%02X", 
          mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  
  return String(macStr);
}
